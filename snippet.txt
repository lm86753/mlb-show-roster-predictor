from src.api.main import app
from fastapi.testclient import TestClient

c = TestClient(app)
r = c.get('/dashboard?horizon_days=1')
d = r.json()
print('count', d['count'])
p = d['predictions'][0]
print('keys:', sorted(p.keys()))
print()
print('=== top 5 by predicted_ovr_delta ===')
top = sorted(d['predictions'], key=lambda x: x.get('predicted_ovr_delta', 0) or 0, reverse=True)[:5]
for pp in top:
    print(f"{pp['player_name']:25} {pp['current_ovr']} {pp['current_rarity']:10} P{pp.get('upgrade_probability')} D{pp.get('downgrade_probability')} I{pp.get('investment_score')} EV{pp.get('expected_value_per_card')}")
print()
print('update_status:', d['update_status'])
